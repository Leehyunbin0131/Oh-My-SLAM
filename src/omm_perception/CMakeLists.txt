cmake_minimum_required(VERSION 3.8)
project(omm_perception)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
foreach(dep rclcpp sensor_msgs geometry_msgs std_srvs tf2_ros tf2_geometry_msgs cv_bridge omm_perception_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(memory_core SHARED src/memory.cpp src/ground.cpp src/tracker.cpp)
target_include_directories(memory_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${OpenCV_INCLUDE_DIRS})
target_link_libraries(memory_core Eigen3::Eigen SQLite::SQLite3 ${OpenCV_LIBRARIES})
add_library(obstacle_memory_node SHARED src/obstacle_memory_node.cpp)
target_link_libraries(obstacle_memory_node memory_core)
ament_target_dependencies(obstacle_memory_node rclcpp sensor_msgs geometry_msgs std_srvs tf2_ros tf2_geometry_msgs cv_bridge omm_perception_msgs)
add_executable(obstacle_memory src/main.cpp)
target_link_libraries(obstacle_memory obstacle_memory_node)
ament_target_dependencies(obstacle_memory rclcpp)
install(TARGETS memory_core obstacle_memory_node LIBRARY DESTINATION lib)
install(TARGETS obstacle_memory DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(rosgraph_msgs REQUIRED)
  find_package(tf2_msgs REQUIRED)
  ament_add_gtest(test_memory test/test_memory.cpp)
  target_link_libraries(test_memory memory_core)
  ament_add_gtest(test_geometry test/test_geometry.cpp test/test_session.cpp ENV ROS_DOMAIN_ID=91)
  target_link_libraries(test_geometry memory_core obstacle_memory_node)
  ament_target_dependencies(test_geometry rclcpp rosgraph_msgs tf2_msgs sensor_msgs omm_perception_msgs)
endif()
install(FILES LICENSE provenance.json DESTINATION share/${PROJECT_NAME})
ament_package()
