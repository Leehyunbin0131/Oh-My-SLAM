cmake_minimum_required(VERSION 3.8)
project(omm_costmap_layers)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep nav2_costmap_2d nav_msgs pluginlib rclcpp tf2_geometry_msgs omm_perception_msgs omslam_operator_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(obstacle_memory_layer SHARED src/obstacle_memory_layer.cpp src/keepout_layer.cpp)
target_include_directories(obstacle_memory_layer PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(obstacle_memory_layer nav2_costmap_2d nav_msgs pluginlib rclcpp tf2_geometry_msgs omm_perception_msgs omslam_operator_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS obstacle_memory_layer LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES LICENSE provenance.json DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_memory_layer_projection test/test_memory_layer_projection.cpp ENV ROS_DOMAIN_ID=92)
  target_link_libraries(test_memory_layer_projection obstacle_memory_layer)
  ament_target_dependencies(test_memory_layer_projection nav2_costmap_2d rclcpp tf2_geometry_msgs omm_perception_msgs omslam_operator_msgs)
  ament_add_gtest(test_keepout_layer test/test_keepout_layer.cpp ENV ROS_DOMAIN_ID=93)
  target_link_libraries(test_keepout_layer obstacle_memory_layer)
  ament_target_dependencies(test_keepout_layer nav2_costmap_2d rclcpp tf2_geometry_msgs omslam_operator_msgs)
endif()
ament_package()
